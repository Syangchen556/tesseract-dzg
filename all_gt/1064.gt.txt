སོ་མཱ་ལི་ཡ་རྒྱལ་ཁབ་འདི་ སྤྱི་མཐུན་སྤྱི་ཚོགས་ཀྱི་སྤྱི་མཐུན་རྒྱལ་ཁབ་ཅིག་སྦེ་ལཱ་འབདཝ་ཨིན་མས། སོ་མཱ་ལི་ཡའི་སྤྱི་མཐུན་གཞུང་འདི་ བཀོད་ཁྱབ་ དང་ཁྲིམས་བཟོ་ དེ་ལས་ ཁྲིམས་འདུན་གྱི་ཡན་ལག་གསུམ་ལུ་དབྱེ་སྟེ་ཡོདཔ་ཨིན་མས། སྲིད་འཛིན་གྱིས་ རྒྱལ་ཁབ་ཀྱི་དབུ་ཁྲིདཔ་སྦེ་ ཕྱག་ཞུ་དོ་ ཡོདཔ་ད་ བློན་ཆེན་གྱིས་ གཞུང་གི་དབུ་ཁྲིདཔ་སྦེ་ ཕྱག་ཞུ་དོ་ཡོདཔ་ཨིན་པས། སྤྱི་ཚོགས་འདི་ མི་སེར་གྱི་ཚོགས་ཁང་དང་ ཚོགས་ཁང་ གོང་མ་འབད་ གཉིས་ཡོདཔ་ཨིན་མས།
