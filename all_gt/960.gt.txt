རྒྱལ་ཁབ་འདི་ བདག་སྐྱོང་གི་ས་ཁོངས་ལྔ་དང་ དེ་གུ་ རྒྱལ་ས་ལྟེ་བ་བྷཱན་ཇུལ་གྱི་ཁྲོམ་སྡེ་བརྩིས་ཏེ་ བདག་སྐྱོང་ས་ཁོངས་ ༦ ལུ་ཁག་དབྱེ་སྟེ་ཡོདཔ་ཨིན། རྒྱལ་ཁབ་འདི་ མང་ཤོས་རང་ གཙང་ཆུ་དང་ ས་བྷ་ནའི་རྩ་ཐང་ཚུ་གིས་ཁྱབ་སྟེ་ཡོདཔ་ཨིན། རྒྱལ་ཁབ་འདིའི་གནམ་གཤིས་གནས་སྟངས་ཡང་ མང་ཤོས་རང་ཚ་དྲོད་རན་ཏོག་ཏོ་སྦེ་སྡོད་དོ་ཡོདཔ་ཨིན། དེ་ཡང་ དབྱིན་ཟླ་༡༡ པ་ལས་དབྱིན་ཟླ་༥ པ་ཚུན་ ཚ་ཏོག་ཏོ་དང་སྐམ་ཀྲོག་ཀྲོ་སྦེ་སྡོད་དོ་ཡོདཔ་དང་ དབྱིན་ཟླ་༦ པ་དང་༡༠ པའི་བར་ན་ ཆརཔ་རྐྱབ་དོ་ཡོདཔ་ཨིན།
