སྤྱི་ལོ་ ༡༩༡༢ སྤྱི་ཟླ་ ༡༡ པའི་ཚེས་ ༢༨ ལུ་ ཨཱལ་བེ་ནི་ཡ་གིས་ ཨོ་ཊོ་མཱན་རྒྱལ་ཁབ་ལས་ རང་དབང་ཐོབ་པའི་གསལ་བསྒྲགས་འབད་དེ་ དེང་སང་གི་དུས་སྐབས་འགོ་བཙུགས་ནུག། སྤྱི་ལོ་ ༢༠༢༣ ལོའི་མི་རྩིས་ལྟར་འབད་བ་ཅིན་ ཨཱལ་བྷེ་ནི་ཡ་ལུ་ མི་རློབས་ས་ཡ་༢༤ དེ་ཅིག་ཡོདཔ་ཨིན། མི་མང་ཤོས་ཅིག་གིས་ར་ ཨལ་བྷེ་ནི་ཡ་གི་མི་རིགས་ཨིནམ་སྦེ་ ངོས་འཛིན་འབད་དོ་ཡོདཔ་དང་ འདི་ཡང་ གིརིཀ་དང་ མེག་ཌོ་ནི་ཡཱན་ དེ་ལས་ མོན་ཊེན་གིརིན་དང་རོ་མཱ་ཚུ་བརྩིས་ཏེ་ མི་རིགས་གྱངས་ཁ་ཆོད་མི་ཚུ་ཨིན་མས། ཨལ་བྷེ་ནི་ཡ ལུ་ མི་རིགས་ཀྱི་ལམ་ལུགས་མེདཔ་ཨིན། མི་སྡེ་གི་གཞི་བཀོད་འདི་ ལུང་ཕྱོགས་དང་བཟའ་ཚང་གི་འབྲེལ་བ་ཚུ་གིས་ ཤན་ཞུགས་འབད་དེ་ མི་སྡེ་དང་མགྲོན་བསུ་འབད་ནི་ལུ་ གཙོ་བོར་བསྟེནམ་ཨིན་མས།
